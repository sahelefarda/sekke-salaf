حالت دوم: رشد سکه در ۵۹ روز گذشته (S_now > K)

اگر سکه در ۵۹ روز گذشته بیش از ۷٪ نسبت به P₀ رشد کرده باشد، اختیار فروش تبعی دولت وارد ناحیه خارج از سود (OTM) شده است. یعنی اگر شما امروز عسکه را بالای K بخرید و سکه ریزش کند، دولت فقط تا K را جبران می‌کند و مابقی افت به عهده خریدار امروز است.

قاعده طلایی آربیتراژ جایگزینی (Replacement Rule)

حتی زمانی که S_now > K باشد، تا وقتی که قیمت خرید عسکه در تابلو (Ask_Asakeh) کمتر یا مساوی قیمت فروش گواهی سکه نقدی (Bid_Spot) به علاوه ارزش زمانی اختیار تبعی باشد، تبدیل سکه نقدی به عسکه کاملاً به نفع معامله‌گر است.

اسکریپت پایتون ارزیابی «عسکه» با فرمول اصلاح‌شده K = 1.07 × P₀ و دریافت زنده از TSETMC

شامل نرمال‌سازی حروف فارسی، دریافت مظنه Bid/Ask از TSETMC و محاسبه اختیار فروش تبعی برای ۳۱ روز باقیمانده

import time
import math
import requests
from scipy.stats import norm

# ---------------------------------------------------------------------------
# TSE / IME Quantitative Engine: Asakeh Parallel Forward vs Spot Coin
# Total Tenure: 90 Days | Elapsed: 59 Days | Remaining: 31 Days (18 Azar 1405)
# Guaranteed Floor at Maturity: K = P0_initial * (1 + 0.07)
# ---------------------------------------------------------------------------

HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/122.0.0.0 Safari/537.36",
    "Accept": "application/json, text/plain, */*",
}

def normalize_persian(text: str) -> str:
    """نرمال‌سازی حروف ی و ک عربی به فارسی استاندارد برای جستجوی نماد در بورس تهران"""
    return text.replace("ي", "ی").replace("ك", "ک").strip()

def fetch_tsetmc_orderbook(ins_code: str, retries: int = 3) -> dict:
    """دریافت مظنه‌های Best Bid و Best Ask از API رسمی TSETMC با مکانیزم ضد مسدودی"""
    url_price = f"https://cdn.tsetmc.com/api/ClosingPrice/GetClosingPriceInfo/{ins_code}"
    url_ob = f"https://cdn.tsetmc.com/api/BestLimits/{ins_code}"
    
    for attempt in range(retries):
        try:
            rp = requests.get(url_price, headers=HEADERS, timeout=7)
            rob = requests.get(url_ob, headers=HEADERS, timeout=7)
            rp.raise_for_status()
            rob.raise_for_status()
            
            p_info = rp.json().get("closingPriceInfo", {})
            limits = rob.json().get("bestLimits", [])
            best_limit = next((row for row in limits if row.get("number") == 1), {})
            
            return {
                "last": float(p_info.get("pDrCotVal", 0)) / 10.0,   # تبدیل ریال به تومان
                "close": float(p_info.get("pClosing", 0)) / 10.0,
                "bid": float(best_limit.get("pMeDem", 0)) / 10.0,
                "ask": float(best_limit.get("pMeOf", 0)) / 10.0,
            }
        except Exception:
            time.sleep(1.5 * (attempt + 1))
    raise RuntimeError(f"Failed to fetch TSETMC orderbook for {ins_code}")

def evaluate_asakeh_contract(
    P0_initial: float,
    S_now: float,
    asakeh_ask: float,
    spot_bid: float,
    guar_rate: float = 0.07,
    days_rem: int = 31,
    r_rf: float = 0.31,
    sigma: float = 0.35
) -> dict:
    """
    محاسبه ارزش ذاتی عسکه بر مبنای تضمین ۷٪ نسبت به قیمت عرضه اولیه ۵۹ روز پیش (P0)
    K = P0_initial * (1 + guar_rate)
    V_asakeh = S_now + Put_BS(S_now, K, T=31/365, r_rf, sigma)
    """
    K_floor = P0_initial * (1.0 + guar_rate)
    T = days_rem / 365.0
    
    d1 = (math.log(S_now / K_floor) + (r_rf + 0.5 * sigma ** 2) * T) / (sigma * math.sqrt(T))
    d2 = d1 - sigma * math.sqrt(T)
    
    put_val = K_floor * math.exp(-r_rf * T) * norm.cdf(-d2) - S_now * norm.cdf(-d1)
    intrinsic_put = max(0.0, K_floor - S_now)
    time_val_put = max(0.0, put_val - intrinsic_put)
    
    fair_value = S_now + put_val
    worst_case_return_pct = ((K_floor - asakeh_ask) / asakeh_ask) * 100.0
    
    return {
        "K_floor": K_floor,
        "put_total": put_val,
        "put_intrinsic": intrinsic_put,
        "put_time_value": time_val_put,
        "fair_value": fair_value,
        "net_alpha": fair_value - asakeh_ask,
        "worst_case_return_pct": worst_case_return_pct,
    }

if __name__ == "__main__":
    # مقادیر نمونه به تومان برای معادل ۱ سکه کامل (۱۰۰۰ ورقه عسکه)
    res = evaluate_asakeh_contract(
        P0_initial=240_000_000,   # قیمت عرضه اولیه ۵۹ روز پیش
        S_now=256_000_000,        # قیمت لحظه‌ای سکه در بورس کالا
        asakeh_ask=249_200_000,   # بهترین مظنه فروش ۱۰۰۰ ورقه عسکه امروز
        spot_bid=255_600_000      # بهترین مظنه خرید گواهی سکه امروز
    )
    print(f"کف تعهد دولت در سررسید (K = 1.07 * P0): {res['K_floor']:,.0f} تومان")
    print(f"ارزش اختیار فروش تبعی دولت (Put): {res['put_total']:,.0f} تومان")
    print(f"ارزش ذاتی امروز ۱۰۰۰ عسکه: {res['fair_value']:,.0f} تومان")
    print(f"حداقل بازده قطعی خریدار امروز در بدترین ریزش سکه: {res['worst_case_return_pct']:.2f}%")
